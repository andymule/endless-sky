cmake_minimum_required(VERSION 3.24)
project(imgui_test VERSION 0.1.0)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)

# Add warning flags
if(MSVC)
    add_compile_options(/W4)
    add_compile_options(/wd4100) # Unused parameter
    add_compile_options(/wd4505) # Unreferenced local function
else()
    add_compile_options(-Wall -Wextra)
    add_compile_options(-Wno-unused-parameter)
    add_compile_options(-Wno-unused-variable)
endif()

# Add ImGui and OpenGL specific definitions
add_definitions(
    -DIMGUI_DISABLE_OBSOLETE_FUNCTIONS
    -DIMGUI_DISABLE_OBSOLETE_KEYIO
    -DIMGUI_IMPL_OPENGL_LOADER_GL3W
    -DIMGUI_IMPL_OPENGL_USE_DESKTOP_GL
)

# Add the executable target
add_executable(imgui_test test_imgui_simple.cpp)

# Find or fetch ImGui
include(FetchContent)

# Fetch ImGui
FetchContent_Declare(
    imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG v1.90.1
)
FetchContent_MakeAvailable(imgui)

# Fetch gl3w
FetchContent_Declare(
    gl3w
    GIT_REPOSITORY https://github.com/skaslev/gl3w.git
    GIT_TAG master
)
FetchContent_MakeAvailable(gl3w)

# Add ImGui sources
set(IMGUI_SOURCES
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_demo.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp
)

# Add gl3w sources
set(GL3W_SOURCES
    ${gl3w_SOURCE_DIR}/src/gl3w.c
)

target_sources(imgui_test PRIVATE ${IMGUI_SOURCES} ${GL3W_SOURCES})

# Find SDL2 using pkg-config
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)

# If pkg-config fails, fall back to traditional find_package
if(NOT SDL2_FOUND)
    find_package(SDL2 REQUIRED)
    if(NOT SDL2_FOUND)
        message(FATAL_ERROR "SDL2 not found.")
    endif()
endif()

# Find OpenGL
find_package(OpenGL REQUIRED)

# Include directories
target_include_directories(imgui_test PRIVATE
    ${imgui_SOURCE_DIR}
    ${imgui_SOURCE_DIR}/backends
    ${gl3w_SOURCE_DIR}/include
    ${SDL2_INCLUDE_DIRS}
)

# Link with SDL2 and OpenGL
target_link_libraries(imgui_test PRIVATE
    ${SDL2_LIBRARIES}
    OpenGL::GL
)

# Add SDL2 library directories and link flags from pkg-config
if(SDL2_FOUND AND SDL2_LINK_LIBRARIES)
    target_link_directories(imgui_test PRIVATE ${SDL2_LIBRARY_DIRS})
    target_link_libraries(imgui_test PRIVATE ${SDL2_LINK_LIBRARIES})
endif()

# Copy required DLLs to build directory (Windows)
if(WIN32)
    find_file(SDL2_DLL
        NAMES SDL2.dll
        PATHS ${SDL2_LIBRARY_DIRS} "C:/msys64/mingw64/bin"
        NO_DEFAULT_PATH
    )
    if(SDL2_DLL)
        add_custom_command(TARGET imgui_test POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDL2_DLL}" "$<TARGET_FILE_DIR:imgui_test>/"
            COMMENT "Copying SDL2.dll to build directory"
        )
    else()
        message(WARNING "SDL2.dll not found in ${SDL2_LIBRARY_DIRS} or C:/msys64/mingw64/bin")
    endif()
endif()
